% Compact black-and-white PathoAlign architecture diagram.
% Two shallow rows keep every label readable at full paper width.
\begin{tikzpicture}[
  x=1cm,
  y=1cm,
  >=Latex,
  block/.style={draw=black, line width=0.78pt, fill=white, rounded corners=1.5pt, align=center, minimum height=0.88cm, inner sep=5pt, font=\small},
  branch/.style={draw=black, line width=0.95pt, fill=white, rounded corners=2pt, align=center, minimum height=1.00cm, inner sep=5pt, font=\small},
  objective/.style={draw=black, line width=0.62pt, densely dashed, fill=white, rounded corners=1pt, align=center, minimum height=0.92cm, inner sep=4pt, font=\scriptsize},
  audit/.style={draw=black, line width=0.72pt, fill=white, align=center, minimum height=0.92cm, inner sep=4pt, font=\scriptsize},
  arrow/.style={->, draw=black, line width=0.85pt},
  dashedarrow/.style={->, draw=black, line width=0.60pt, densely dashed},
  note/.style={font=\scriptsize, align=center}
]

% Shared input and encoder.
\node[block, minimum width=2.60cm] (views) at (1.30,1.42) {same tissue region\\five scanner views\\$\{x_{r,s}\}_{s=1}^{5}$};
\node[note] at (1.30,0.53) {biology fixed; acquisition changes};

\node[block, minimum width=2.20cm] (encoder) at (4.30,1.42) {frozen encoder\\ResNet50, DINOv2,\\or Phikon};
\draw[arrow] (views.east) -- (encoder.west);

% Factor branches.
\node[branch, minimum width=2.25cm] (bio) at (7.25,2.18) {biological branch\\$z_b\in\mathbb{R}^{256}$};
\node[branch, minimum width=2.25cm] (acq) at (7.25,0.65) {acquisition branch\\$z_a\in\mathbb{R}^{64}$};
\draw[arrow] (encoder.east) -- ++(0.42,0) |- (bio.west);
\draw[arrow] (encoder.east) -- ++(0.42,0) |- (acq.west);

% Branch-specific objectives.
\node[objective, minimum width=2.75cm] (bioobj) at (10.35,2.18) {same-region contrast\\scanner dependence penalty\\gradient reversal};
\node[objective, minimum width=2.75cm] (acqobj) at (10.35,0.65) {scanner prediction\\from acquisition factor};
\draw[dashedarrow] (bio.east) -- (bioobj.west);
\draw[dashedarrow] (acq.east) -- (acqobj.west);

% Output audits.
\node[audit, minimum width=2.55cm] (bioaudit) at (13.55,2.18) {tissue retrieval preserved\\scanner probe reduced};
\node[audit, minimum width=2.55cm] (acqaudit) at (13.55,0.65) {scanner identity retained\\tissue retrieval reduced};
\draw[arrow] (bioobj.east) -- (bioaudit.west);
\draw[arrow] (acqobj.east) -- (acqaudit.west);

% Joint constraints below both branches.
\node[objective, minimum width=5.05cm, minimum height=0.72cm] (joint) at (9.15,-0.72) {joint reconstruction $\widehat{x}=g([z_b,z_a])$ \quad + \quad variance, covariance, and cross-factor decorrelation};
\draw[dashedarrow] (bio.south) -- ++(0,-0.28) -| (joint.north west);
\draw[dashedarrow] (acq.south) -- ++(0,-0.20) -| (joint.north east);

\node[note, anchor=west] at (11.85,-0.70) {separate the factors; do not erase the representation};
\end{tikzpicture}
