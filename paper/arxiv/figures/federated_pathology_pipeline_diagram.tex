% #1 x, #2 y, #3 width, #4 height, #5 depth-x, #6 depth-y,
% #7 base colour, #8 label, #9 coordinate name.
\newcommand{\Cuboid}[9]{%
  \pgfmathsetmacro{\xa}{#1}%
  \pgfmathsetmacro{\ya}{#2}%
  \pgfmathsetmacro{\xb}{#1+#3}%
  \pgfmathsetmacro{\yb}{#2+#4}%
  \pgfmathsetmacro{\xd}{#1+#5}%
  \pgfmathsetmacro{\yd}{#2+#6}%
  \pgfmathsetmacro{\xbd}{#1+#3+#5}%
  \pgfmathsetmacro{\ybd}{#2+#4+#6}%
  \filldraw[fill=#7!90,draw=black!72,line width=0.55pt]
    (\xa,\ya) rectangle (\xb,\yb);
  \filldraw[fill=#7!62,draw=black!72,line width=0.55pt]
    (\xb,\ya) -- (\xbd,\yd) -- (\xbd,\ybd) -- (\xb,\yb) -- cycle;
  \filldraw[fill=#7!42,draw=black!72,line width=0.55pt]
    (\xa,\yb) -- (\xd,\ybd) -- (\xbd,\ybd) -- (\xb,\yb) -- cycle;
  \node[align=center,font=\scriptsize,text=black] at ({(\xa+\xb)/2},{(\ya+\yb)/2}) {#8};
  \coordinate (#9-west) at (\xa,{(\ya+\yb)/2});
  \coordinate (#9-east) at (\xbd,{(\yd+\ybd)/2});
  \coordinate (#9-north) at ({(\xd+\xbd)/2},\ybd);
  \coordinate (#9-south) at ({(\xa+\xb)/2},\ya);
  \coordinate (#9-center) at ({(\xa+\xb)/2},{(\ya+\yb)/2});
}

% Stacked cuboid: two offset back plates plus one labelled front plate.
\newcommand{\StackCuboid}[9]{%
  \pgfmathsetmacro{\sx}{#1+0.18}%
  \pgfmathsetmacro{\sy}{#2+0.16}%
  \pgfmathsetmacro{\sxx}{#1+0.36}%
  \pgfmathsetmacro{\syy}{#2+0.32}%
  \Cuboid{\sxx}{\syy}{#3}{#4}{#5}{#6}{#7}{}{#9-backtwo}%
  \Cuboid{\sx}{\sy}{#3}{#4}{#5}{#6}{#7}{}{#9-backone}%
  \Cuboid{#1}{#2}{#3}{#4}{#5}{#6}{#7}{#8}{#9}%
}
