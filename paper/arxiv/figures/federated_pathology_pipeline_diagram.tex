% Publication-readable wireframe schematic for federated computational pathology.
% The shared local architecture is shown once; client heterogeneity and updates
% are represented separately to avoid repeating the entire network three times.

\newcommand{\Stack}[7]{%
  % x, y, w, h, depth, label, name
  \foreach \i in {#5,...,1}{%
    \pgfmathsetmacro{\dx}{0.09*\i}%
    \pgfmathsetmacro{\dy}{0.07*\i}%
    \draw[thinwire] ({#1+\dx},{#2+\dy}) rectangle ({#1+#3+\dx},{#2+#4+\dy});
  }%
  \draw[wire] (#1,#2) rectangle ({#1+#3},{#2+#4});
  \draw[thinwire] (#1,{#2+#4}) -- ({#1+0.09*#5},{#2+#4+0.07*#5});
  \draw[thinwire] ({#1+#3},{#2+#4}) -- ({#1+#3+0.09*#5},{#2+#4+0.07*#5});
  \draw[thinwire] ({#1+#3},#2) -- ({#1+#3+0.09*#5},{#2+0.07*#5});
  \node[inside,text width={#3 cm}] at ({#1+#3/2},{#2+#4/2}) {#6};
  \coordinate (#7-west) at (#1,{#2+#4/2});
  \coordinate (#7-east) at ({#1+#3+0.09*#5},{#2+#4/2+0.07*#5});
  \coordinate (#7-north) at ({#1+#3/2},{#2+#4+0.07*#5});
  \coordinate (#7-south) at ({#1+#3/2},#2);
}

\newcommand{\Vector}[6]{%
  % x, y, w, h, label, name
  \draw[wire] (#1,#2) rectangle ({#1+#3},{#2+#4});
  \draw[thinwire] ({#1+0.08},{#2+0.06}) rectangle ({#1+#3+0.08},{#2+#4+0.06});
  \node[inside,text width={#3 cm}] at ({#1+#3/2},{#2+#4/2}) {#5};
  \coordinate (#6-west) at (#1,{#2+#4/2});
  \coordinate (#6-east) at ({#1+#3+0.08},{#2+#4/2+0.06});
  \coordinate (#6-north) at ({#1+#3/2},{#2+#4+0.06});
  \coordinate (#6-south) at ({#1+#3/2},#2);
}
