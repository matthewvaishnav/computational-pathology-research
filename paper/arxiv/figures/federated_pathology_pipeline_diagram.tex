% Compact architecture-and-mathematics figure for federated computational pathology.
% The local model is shown as a short tensor pipeline; the server-side mechanism
% is expressed directly with the aggregation and detector equations.

\newcommand{\TensorBlock}[7]{%
  \foreach \i in {#5,...,1}{%
    \pgfmathsetmacro{\dx}{0.075*\i}%
    \pgfmathsetmacro{\dy}{0.055*\i}%
    \draw[tensorback] ({#1+\dx},{#2+\dy}) rectangle ({#1+#3+\dx},{#2+#4+\dy});
  }%
  \draw[tensorfront] (#1,#2) rectangle ({#1+#3},{#2+#4});
  \draw[tensorback] (#1,{#2+#4}) -- ({#1+0.075*#5},{#2+#4+0.055*#5});
  \draw[tensorback] ({#1+#3},{#2+#4}) -- ({#1+#3+0.075*#5},{#2+#4+0.055*#5});
  \draw[tensorback] ({#1+#3},#2) -- ({#1+#3+0.075*#5},{#2+0.055*#5});
  \node[tensorlabel,text width={#3 cm}] at ({#1+#3/2},{#2+#4/2}) {#6};
  \coordinate (#7-west) at (#1,{#2+#4/2});
  \coordinate (#7-east) at ({#1+#3+0.075*#5},{#2+#4/2+0.055*#5});
}

\newcommand{\ThinLayer}[6]{%
  \draw[tensorfront] (#1,#2) rectangle ({#1+#3},{#2+#4});
  \draw[tensorback] ({#1+0.06},{#2+0.045}) rectangle ({#1+#3+0.06},{#2+#4+0.045});
  \node[tensorlabel,text width={#3 cm}] at ({#1+#3/2},{#2+#4/2}) {#5};
  \coordinate (#6-west) at (#1,{#2+#4/2});
  \coordinate (#6-east) at ({#1+#3+0.06},{#2+#4/2+0.045});
}
