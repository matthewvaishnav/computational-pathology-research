% Draw a perspective stack.
% #1 x, #2 y, #3 width, #4 height, #5 number of visible back planes,
% #6 horizontal offset, #7 vertical offset, #8 label, #9 coordinate name.
\newcommand{\WireStack}[9]{%
  \pgfmathsetmacro{\wx}{#1}%
  \pgfmathsetmacro{\wy}{#2}%
  \pgfmathsetmacro{\ww}{#3}%
  \pgfmathsetmacro{\wh}{#4}%
  \foreach \i in {#5,...,1}{%
    \pgfmathsetmacro{\ox}{\i*(#6)}%
    \pgfmathsetmacro{\oy}{\i*(#7)}%
    \draw[wirethin]
      ({\wx+\ox},{\wy+\oy}) --
      ({\wx+\ww+\ox},{\wy+\oy}) --
      ({\wx+\ww+\ox},{\wy+\wh+\oy}) --
      ({\wx+\ox},{\wy+\wh+\oy}) -- cycle;
  }%
  \draw[wire]
    (\wx,\wy) -- (\wx+\ww,\wy) --
    (\wx+\ww,\wy+\wh) -- (\wx,\wy+\wh) -- cycle;
  \draw[wirethin] (\wx,\wy+\wh) -- ({\wx+#5*(#6)},{\wy+\wh+#5*(#7)});
  \draw[wirethin] (\wx+\ww,\wy+\wh) -- ({\wx+\ww+#5*(#6)},{\wy+\wh+#5*(#7)});
  \draw[wirethin] (\wx+\ww,\wy) -- ({\wx+\ww+#5*(#6)},{\wy+#5*(#7)});
  \node[blocklabel,text width={#3 cm}] at ({\wx+\ww/2},{\wy+\wh/2}) {#8};
  \coordinate (#9-west) at (\wx,{\wy+\wh/2});
  \coordinate (#9-east) at ({\wx+\ww+#5*(#6)},{\wy+\wh/2+#5*(#7)});
  \coordinate (#9-north) at ({\wx+\ww/2},{\wy+\wh+#5*(#7)});
  \coordinate (#9-south) at ({\wx+\ww/2},\wy);
}

% Draw a narrow dense layer.
% #1 x, #2 y, #3 height, #4 label, #5 coordinate name.
\newcommand{\DenseLayer}[5]{%
  \draw[wire] (#1,#2) rectangle ({#1+0.34},{#2+#3});
  \draw[wirethin] ({#1+0.10},{#2+0.08}) rectangle ({#1+0.44},{#2+#3+0.08});
  \node[rotate=90,font=\scriptsize] at ({#1+0.17},{#2+#3/2}) {#4};
  \coordinate (#5-west) at (#1,{#2+#3/2});
  \coordinate (#5-east) at ({#1+0.44},{#2+#3/2+0.08});
  \coordinate (#5-south) at ({#1+0.17},#2);
}
