% Compact publication-readable wireframe architecture for federated pathology.
% Designed to remain legible when resized to a single-column text width.

\newcommand{\Stack}[7]{%
  % x, y, width, height, depth, label, name
  \foreach \i in {#5,...,1}{%
    \pgfmathsetmacro{\dx}{0.075*\i}%
    \pgfmathsetmacro{\dy}{0.055*\i}%
    \draw[thinwire] ({#1+\dx},{#2+\dy}) rectangle ({#1+#3+\dx},{#2+#4+\dy});
  }%
  \draw[wire] (#1,#2) rectangle ({#1+#3},{#2+#4});
  \draw[thinwire] (#1,{#2+#4}) -- ({#1+0.075*#5},{#2+#4+0.055*#5});
  \draw[thinwire] ({#1+#3},{#2+#4}) -- ({#1+#3+0.075*#5},{#2+#4+0.055*#5});
  \draw[thinwire] ({#1+#3},#2) -- ({#1+#3+0.075*#5},{#2+0.055*#5});
  \node[inside,text width={#3 cm}] at ({#1+#3/2},{#2+#4/2}) {#6};
  \coordinate (#7-west) at (#1,{#2+#4/2});
  \coordinate (#7-east) at ({#1+#3+0.075*#5},{#2+#4/2+0.055*#5});
  \coordinate (#7-north) at ({#1+#3/2},{#2+#4+0.055*#5});
  \coordinate (#7-south) at ({#1+#3/2},#2);
}

\newcommand{\Vector}[6]{%
  % x, y, width, height, label, name
  \draw[wire] (#1,#2) rectangle ({#1+#3},{#2+#4});
  \draw[thinwire] ({#1+0.065},{#2+0.05}) rectangle ({#1+#3+0.065},{#2+#4+0.05});
  \node[inside,text width={#3 cm}] at ({#1+#3/2},{#2+#4/2}) {#5};
  \coordinate (#6-west) at (#1,{#2+#4/2});
  \coordinate (#6-east) at ({#1+#3+0.065},{#2+#4/2+0.05});
  \coordinate (#6-north) at ({#1+#3/2},{#2+#4+0.05});
  \coordinate (#6-south) at ({#1+#3/2},#2);
}
