% Readable black-and-white wireframe architecture for federated computational pathology.

% Perspective wireframe block.
% #1 x, #2 y, #3 width, #4 height, #5 depth count,
% #6 label, #7 coordinate name.
\newcommand{\WireBlock}[7]{%
  \foreach \i in {#5,...,1}{%
    \pgfmathsetmacro{\dx}{0.08*\i}%
    \pgfmathsetmacro{\dy}{0.06*\i}%
    \draw[thinwire]
      ({#1+\dx},{#2+\dy}) rectangle
      ({#1+#3+\dx},{#2+#4+\dy});
  }%
  \draw[wire] (#1,#2) rectangle ({#1+#3},{#2+#4});
  \draw[thinwire] (#1,{#2+#4}) -- ({#1+0.08*#5},{#2+#4+0.06*#5});
  \draw[thinwire] ({#1+#3},{#2+#4}) -- ({#1+#3+0.08*#5},{#2+#4+0.06*#5});
  \draw[thinwire] ({#1+#3},#2) -- ({#1+#3+0.08*#5},{#2+0.06*#5});
  \node[inside,text width={#3 cm}] at ({#1+#3/2},{#2+#4/2}) {#6};
  \coordinate (#7-west) at (#1,{#2+#4/2});
  \coordinate (#7-east) at ({#1+#3+0.08*#5},{#2+#4/2+0.06*#5});
  \coordinate (#7-north) at ({#1+#3/2},{#2+#4+0.06*#5});
  \coordinate (#7-south) at ({#1+#3/2},#2);
}

% Narrow vector/dense layer.
% #1 x, #2 y, #3 width, #4 height, #5 label, #6 coordinate name.
\newcommand{\VectorLayer}[6]{%
  \draw[wire] (#1,#2) rectangle ({#1+#3},{#2+#4});
  \draw[thinwire] ({#1+0.08},{#2+0.06}) rectangle ({#1+#3+0.08},{#2+#4+0.06});
  \node[inside,text width={#3 cm}] at ({#1+#3/2},{#2+#4/2}) {#5};
  \coordinate (#6-west) at (#1,{#2+#4/2});
  \coordinate (#6-east) at ({#1+#3+0.08},{#2+#4/2+0.06});
  \coordinate (#6-north) at ({#1+#3/2},{#2+#4+0.06});
  \coordinate (#6-south) at ({#1+#3/2},#2);
}
